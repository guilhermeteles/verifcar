Uso: Títulos, headings, nomes de seções, KPI values, logo text

Flutter ThemeData
Tokens organizados para integração direta com Flutter. Copie o bloco completo.
// ─── VerifCar Cortex Indigo — Flutter Theme ───
import 'package:flutter/material.dart';
import 'package:google_fonts/google_fonts.dart';

class CortexIndigo {
  // ── Primary ──
  static const pri50  = Color(0xFFECEEFF);
  static const pri100 = Color(0xFFD8DAFF);
  static const pri400 = Color(0xFF5C66D4);
  static const pri600 = Color(0xFF3D4592);
  static const pri800 = Color(0xFF252D7A);

  // ── Emerald (Success) ──
  static const em50  = Color(0xFFE6F4E8);
  static const em400 = Color(0xFF1A9E27);
  static const em    = Color(0xFF168821);

  // ── Amber (Warning) ──
  static const am50  = Color(0xFFFDF3DC);
  static const am400 = Color(0xFFC48C14);
  static const am800 = Color(0xFF7A5600);

  // ── Coral (Error) ──
  static const co50  = Color(0xFFFDECEA);
  static const co400 = Color(0xFFE52207);
  static const co800 = Color(0xFF8C1200);

  // ── Surfaces ──
  static const bg      = Color(0xFFF8F9FF);
  static const surface = Colors.white;
  static const surLow  = Color(0xFFF1F3FB);
  static const surMid  = Color(0xFFECEEF6);
  static const surHigh = Color(0xFFE6E8F0);
  static const border  = Color(0xFFD8DAE8);

  // ── Text ──
  static const textH = Color(0xFF181C21);
  static const textB = Color(0xFF464651);
  static const textM = Color(0xFF595C7C);
  static const textS = Color(0xFF767682);

  // ── Spacing ──
  static const sp2  = 2.0;
  static const sp4  = 4.0;
  static const sp6  = 6.0;
  static const sp8  = 8.0;
  static const sp10 = 10.0;
  static const sp12 = 12.0;
  static const sp14 = 14.0;
  static const sp16 = 16.0;
  static const sp20 = 20.0;
  static const sp24 = 24.0;
  static const sp32 = 32.0;

  // ── Border Radius ──
  static final radiusSm   = BorderRadius.circular(6);
  static final radiusMd   = BorderRadius.circular(10);
  static final radiusLg   = BorderRadius.circular(12);
  static final radiusXl   = BorderRadius.circular(14);
  static final radius2xl  = BorderRadius.circular(16);
  static final radiusSheet= BorderRadius.vertical(top: Radius.circular(20));
  static final radiusPill  = BorderRadius.circular(9999);

  // ── Shadows ──
  static final shCard = [
    BoxShadow(color: Color(0x14252D7A), blurRadius: 3, offset: Offset(0,1)),
    BoxShadow(color: Color(0x0A252D7A), blurRadius: 16, offset: Offset(0,4)),
  ];

  // ── ThemeData ──
  static ThemeData get theme => ThemeData(
    scaffoldBackgroundColor: bg,
    colorScheme: ColorScheme.light(
      primary: pri800,
      secondary: pri400,
      error: co400,
      surface: surface,
    ),
    textTheme: TextTheme(
      displayLarge:  GoogleFonts.plusJakartaSans(fontSize: 28, fontWeight: FontWeight.w800, color: pri800),
      titleLarge:    GoogleFonts.plusJakartaSans(fontSize: 22, fontWeight: FontWeight.w800, color: textH),
      titleMedium:   GoogleFonts.plusJakartaSans(fontSize: 15, fontWeight: FontWeight.w700, color: textH),
      titleSmall:    GoogleFonts.plusJakartaSans(fontSize: 13, fontWeight: FontWeight.w700, color: textH),
      bodyLarge:     GoogleFonts.dmSans(fontSize: 14, fontWeight: FontWeight.w400, color: textB),
      bodyMedium:    GoogleFonts.dmSans(fontSize: 12, fontWeight: FontWeight.w500, color: textB),
      bodySmall:     GoogleFonts.dmSans(fontSize: 10, fontWeight: FontWeight.w600, color: textM),
      labelLarge:    GoogleFonts.dmSans(fontSize: 13, fontWeight: FontWeight.w600, color: Colors.white),
      labelSmall:    GoogleFonts.jetBrainsMono(fontSize: 9, fontWeight: FontWeight.w400, color: textS),
    ),
  );
}
